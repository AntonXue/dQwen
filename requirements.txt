# The one environment. Every family (dQwen3.5, LLaDA, Dream, SDAR, CoDA)
# runs here because the comparators are PORTED to transformers 5.x via the
# compat shims in models/*.py -- their native pins are 4.4x-4.5x and none
# of them loads unmodified.
#
# This file is the source of truth for WHAT to install, but it cannot
# cold-build the env: torch+cu128, flash_attn, tilelang and
# flash-linear-attention compile against a specific CUDA toolkit, so the
# base is a hand-built conda env. Install these into it, then run
# setup_env.sh (two source patches to pinned lm-eval; pip cannot express
# them).
#
# torch is pinned EXACTLY: the transformers-5.13 port is bitwise-exact
# against 4.57 only at fixed torch (measured max|d| 4.69e-01 between
# 2.5.1+cu121 and 2.7.1+cu128 on identical weights). torch is part of the
# reproducibility contract, not an incidental dependency.

torch==2.7.1                    # +cu128 build

einops==0.8.1                   # SDAR
datasets==4.4.1
evaluate==0.4.6                 # HF code_eval (the code-benchmark grader)

# The harness. 0.4.8 pinned, not latest: LLaDA and Dream both report

# matching theirs is part of reproducing their numbers. benchmark_specs.py
# is frozen from this pin and tests/task_freeze_gate.py enforces the match.
lm_eval==0.4.8
